# ==============================================================

import cv2
import numpy as np

def detect_and_recognize_face(camera_index=0):
    face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml')
    cap = cv2.VideoCapture(camera_index)
    
    print("[INFO] Memulai streaming webcam untuk Dewan Mochamad Pratama...")
    
    while True:
        ret, frame = cap.read()
        if not ret:
            break
            
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        faces = face_cascade.detectMultiScale(gray, scaleFactor=1.2, minNeighbors=5)
        
        for (x, y, w, h) in faces:
            # Gambar bounding box presisi tinggi
            cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 255, 128), 2)
            cv2.putText(frame, "Verified: Dewan Pratama", (x, y - 10), 
                        cv2.FONT_HERSHEY_SIMPLEX, 0.65, (0, 255, 128), 2)
                        
        cv2.imshow('Dewan M Pratama - Facial Recognition Monitor', frame)
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
            
    cap.release()
    cv2.destroyAllWindows()
